  startDate: string;

  priority: ProgramPriority;
  location: string;

  name: "",

  description: "",
  manager: "",
  beneficiaries: "",
  budget: "",
  spent: "0",
  progress: "0",
  startDate: today(),
  endDate: "",
  status: "Planning",
  priority: "Medium",
  location: "",
};

  isOpen,
  onClose,
  onCreate,

  const [form, setForm] =
    useState<ProgramFormState>(initialForm);

    if (!isOpen) return;

    function handleEscape(event: KeyboardEvent) {
      if (event.key === "Escape") {
        onClose();
      }
    }

    document.body.style.overflow = "hidden";
    document.addEventListener("keydown", handleEscape);

    return () => {
      document.body.style.overflow = "";

    return null;
  }

  function updateField(
    field: keyof ProgramFormState,

      ...current,
      [field]: value,
    }));

  }

  function resetAndClose() {
    setForm(initialForm);
    setError("");
    onClose();
  }

  function handleSubmit(event: FormEvent<HTMLFormElement>) {
    event.preventDefault();

    if (
      !form.name.trim() ||
      !form.manager.trim() ||
      !form.description.trim() ||
      !form.location.trim() ||
      !form.endDate

      return;

      id: generateProgramId(),
      name: form.name.trim(),
      category: form.category,
      description: form.description.trim(),
      manager: form.manager.trim(),

        Number(form.beneficiaries) || 0,
        0
      ),
      budget: Math.max(Number(form.budget) || 0, 0),
      spent: Math.max(Number(form.spent) || 0, 0),
      progress: Math.min(
        Math.max(Number(form.progress) || 0, 0),
        100
      ),
      startDate: form.startDate,
      endDate: form.endDate,
      status: form.status,
      priority: form.priority,
      location: form.location.trim(),
      createdAt: now,
      updatedAt: now,
    };

    onCreate(newProgram);
    setForm(initialForm);
    setError("");
    onClose();
  }

  const inputClassName =
    "mt-2 h-12 w-full rounded-xl border border-white/10 bg-[#0d121b] px-4 text-sm text-white outline-none transition placeholder:text-neutral-600 focus:border-cyan-400";

  search: "",
  status: "All",
  category: "All",
  priority: "All",
};

export default function ProgramsPage() {
  const [programs, setPrograms] =
    useState<Program[]>(PROGRAMS);

  const [filters, setFilters] =
    useState<ProgramFiltersType>(INITIAL_FILTERS);

  const [isStorageReady, setIsStorageReady] =
    useState(false);

  const [isCreateOpen, setIsCreateOpen] =
    useState(false);

  const [viewProgram, setViewProgram] =
    useState<Program | null>(null);

  const [editProgram, setEditProgram] =
    useState<Program | null>(null);

  const [deleteProgram, setDeleteProgram] =
    useState<Program | null>(null);

  useEffect(() => {
    try {
      const storedPrograms =
        window.localStorage.getItem(STORAGE_KEY);

      if (storedPrograms) {
        const parsedPrograms: unknown =
          JSON.parse(storedPrograms);

        if (Array.isArray(parsedPrograms)) {
          setPrograms(parsedPrograms as Program[]);
        }
      }
    } catch (error) {
      console.error(
        "Unable to load saved programs:",
        error
      );
    } finally {
      setIsStorageReady(true);
    }
  }, []);

  useEffect(() => {
    if (!isStorageReady) {
      return;
    }

    try {
      window.localStorage.setItem(
        STORAGE_KEY,
        JSON.stringify(programs)
      );
    } catch (error) {
      console.error(
        "Unable to save programs:",
        error
      );
    }
  }, [programs, isStorageReady]);

    return getVisiblePrograms(programs, filters);

      ...current,
      search: value,
    }));
  }

    setPrograms((current) => [
      program,
      ...current,
    ]);
  }

    setPrograms((current) =>
      current.map((existingProgram) =>
        existingProgram.id === program.id

      )
    );

    setEditProgram(null);
  }

      )
    );

    setDeleteProgram(null);

import { Program, ProgramFilters, ProgramStatistics } from "./types";

/**
 * Search programs by name, category, manager or location
 */
export function searchPrograms(
  programs: Program[],
  search: string
): Program[] {
  if (!search.trim()) return programs;

  const keyword = search.toLowerCase();

  return programs.filter((program) => {
    return (
      program.name.toLowerCase().includes(keyword) ||
      program.category.toLowerCase().includes(keyword) ||
      program.manager.toLowerCase().includes(keyword) ||
      program.location.toLowerCase().includes(keyword)
    );
  });
}

/**
 * Apply filters
 */
export function filterPrograms(
  programs: Program[],
  filters: ProgramFilters
): Program[] {
  return programs.filter((program) => {
    const statusMatch =
      !filters.status ||
      filters.status === "All" ||
      program.status === filters.status;

    const categoryMatch =
      !filters.category ||
      filters.category === "All" ||
      program.category === filters.category;

    const priorityMatch =
      !filters.priority ||
      filters.priority === "All" ||
      program.priority === filters.priority;

    return statusMatch && categoryMatch && priorityMatch;
  });
}

/**
 * Search + Filter together
 */
export function getVisiblePrograms(
  programs: Program[],
  filters: ProgramFilters
): Program[] {
  const searched = searchPrograms(programs, filters.search);

  return filterPrograms(searched, filters);
}

/**
 * Dashboard statistics
 */
export function calculateStatistics(
  programs: Program[]
): ProgramStatistics {
  return {
    active: programs.filter((p) => p.status === "Active").length,

    completed: programs.filter((p) => p.status === "Completed").length,

    inProgress: programs.filter(
      (p) => p.progress > 0 && p.progress < 100
    ).length,

    attention: programs.filter(
      (p) => p.priority === "Critical"
    ).length,

    totalBudget: programs.reduce(
      (sum, p) => sum + p.budget,
      0
    ),

    totalSpent: programs.reduce(
      (sum, p) => sum + p.spent,
      0
    ),

    totalBeneficiaries: programs.reduce(
      (sum, p) => sum + p.beneficiaries,
      0
    ),
  };
}

/**
 * Progress color
 */
export function progressColor(progress: number): string {
  if (progress < 30)
    return "from-red-500 to-orange-400";

  if (progress < 60)
    return "from-yellow-400 to-amber-300";

  if (progress < 90)
    return "from-cyan-400 to-teal-400";

  return "from-emerald-400 to-green-500";
}

/**
 * Status badge color
 */
export function statusColor(status: string): string {
  switch (status) {
    case "Completed":
      return "bg-green-500/20 text-green-400";

    case "Active":
      return "bg-cyan-500/20 text-cyan-400";

    case "Planning":
      return "bg-yellow-500/20 text-yellow-400";

    case "On Hold":
      return "bg-red-500/20 text-red-400";

    default:
      return "bg-neutral-500/20 text-neutral-300";
  }
}

 */

  switch (priority) {
    case "Critical":
      return "text-red-400";

    case "High":
      return "text-orange-400";

    case "Medium":
      return "text-yellow-400";

    default:
      return "text-green-400";
  }
